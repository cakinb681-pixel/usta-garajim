function esc(v){

return String(v??"")
.replace(/[&<>"']/g,function(c){

return {
"&":"&amp;",
"<":"&lt;",
">":"&gt;",
'"':"&quot;",
"'":"&#39;"
}[c];

});

}

function hafizaBul(){

const aranan=
document
.getElementById("hafizaAra")
.value
.toLowerCase()
.trim();

if(!aranan)
return;

const kelimeler=
aranan
.split(/\s+/)
.filter(x=>x.length>2);

const sonuc=
servisler
.map(s=>{

const metin=
(
(s.complaint||"")+" "+
(s.work_done||"")+" "+
(s.parts||"")+" "+
(s.note||"")
).toLowerCase();

let puan=0;

kelimeler.forEach(k=>{

if(metin.includes(k))
puan++;

});

return {...s,puan};

})
.filter(x=>x.puan>0)
.sort((a,b)=>b.puan-a.puan)
.slice(0,10);

document.getElementById("hafizaSonuc")
.innerHTML=
sonuc.map(s=>{

const arac=
araclar.find(x=>x.id===s.tractor_id);

return `
<div class="card">

<b>${esc(arac?.model||"Araç")}</b>

<div class="muted">
${esc(s.service_date)}
</div>

<p>
<b>Arıza:</b>
${esc(s.complaint)}
</p>

<p>
<b>Yapılan:</b>
${esc(s.work_done)}
</p>

<p>
<b>Parçalar:</b>
${esc(s.parts)}
</p>

</div>
`;

}).join("")
||
"<p class='muted'>Benzer kayıt bulunamadı.</p>";

}
